1973 DomainInsideSignature getDomainInside() const{
1974 auto rule = getRule();
1975 if ((rule == TasGrid::rule_gausshermite) || (rule == TasGrid::rule_gausshermiteodd)){ // unbounded domain
1976 return [](std::vector<double> const &)->bool{ return true; };
1977 }else if ((rule == TasGrid::rule_gausslaguerre) || (rule == TasGrid::rule_gausslaguerreodd)){ // bounded from below
1978 if (domain_transform_a.empty()){
1979 return [](std::vector<double> const &x)->bool{
1980 for(auto const &v : x) if (v < 0.0) return false;
1981 return true;
1982 };
1983 }else{
1984 size_t dims = (size_t) getNumDimensions();
1985 return [dta=domain_transform_a, dims](std::vector<double> const &x)->bool{
1986 for(size_t i=0; i<dims; i++) if (x[i] < dta[i]) return false;
1987 return true;
1988 };
1989 }
1990 }else{
1991 if (domain_transform_a.empty()){
1992 if (isFourier()){
1993 return [](std::vector<double> const &x)->bool{
1994 for(auto const &v : x) if ((v < 0.0) || (v > 1.0)) return false;
1995 return true;
1996 };
1997 }else{
1998 return [](std::vector<double> const &x)->bool{
1999 for(auto const &v : x) if ((v < -1.0) || (v > 1.0)) return false;
2000 return true;
2001 };
2002 }
2003 }else{
2004 size_t dims = (size_t) getNumDimensions();
2005 return [dta=domain_transform_a, dtb=domain_transform_b, dims](std::vector<double> const &x)->bool{
2006 for(size_t i=0; i<dims; i++)
2007 if (x[i] < dta[i] or x[i] > dtb[i]) return false;
2008 return true;
2009 };
