147 inline int getNumDimensions() const {return num_dimensions;}
149 inline int getNumParticles() const {return num_particles;}
151 inline void getParticlePositions(double pp[]) const {std::copy_n(particle_positions.begin(), num_particles * num_dimensions, pp);}
153 inline std::vector<double> getParticlePositions() const {return particle_positions;}
155 inline void getParticleVelocities(double pv[]) const {std::copy_n(particle_velocities.begin(), num_particles * num_dimensions, pv);}
157 inline std::vector<double> getParticleVelocities() const {return particle_velocities;}

166 inline std::vector<double> getBestParticlePositions() const {return best_particle_positions;}

170 inline std::vector<double> getBestPosition() const {
171 return std::vector<double>(best_particle_positions.begin() + num_particles * num_dimensions,
172 best_particle_positions.begin() + num_particles * num_dimensions + num_dimensions);
173 }

176 inline bool isPositionInitialized() const {return positions_initialized;}
178 inline bool isVelocityInitialized() const {return velocities_initialized;}
180 inline bool isBestPositionInitialized() const {return best_positions_initialized;}
182 inline bool isCacheInitialized() const {return cache_initialized;}
184 inline std::vector<bool> getStateVector() const {

186 }

189 void setParticlePositions(const double pp[]) {
190 std::copy_n(pp, num_dimensions * num_particles, particle_positions.begin());
191 positions_initialized = true;
192 }
194 void setParticlePositions(const std::vector<double> &pp) {
195 checkVarSize("ParticleSwarmState::setParticlePositions", "particle position", pp.size(), num_dimensions * num_particles);
196 particle_positions = pp;
197 positions_initialized = true;
198 }
200 void setParticlePositions(std::vector<double> &&pp) {
201 checkVarSize("ParticleSwarmState::setParticlePositions", "particle positions", pp.size(), num_dimensions * num_particles);
202 particle_positions = std::move(pp);
203 positions_initialized = true;
204 }
206 void setParticleVelocities(const double pv[]) {
207 std::copy_n(pv, num_dimensions * num_particles, particle_velocities.begin());
208 velocities_initialized = true;
209 }
211 void setParticleVelocities(const std::vector<double> &pv) {
212 checkVarSize("ParticleSwarmState::setParticleVelocities", "particle velocities", pv.size(), num_dimensions * num_particles);
213 particle_velocities = pv;
214 velocities_initialized = true;
215 }
217 void setParticleVelocities(std::vector<double> &&pv) {
218 checkVarSize("ParticleSwarmState::setParticleVelocities", "particle velocities", pv.size(), num_dimensions * num_particles);
219 particle_velocities = std::move(pv);
220 velocities_initialized = true;
221 }
223 void setBestParticlePositions(const double bpp[]) {
224 std::copy_n(bpp, num_dimensions * (num_particles + 1), best_particle_positions.begin());
225 best_positions_initialized = true;
226 }
228 void setBestParticlePositions(const std::vector<double> &bpp) {
229 checkVarSize("ParticleSwarmState::setBestParticlePositions", "best particle positions", bpp.size(), num_dimensions * (num_particles + 1));
230 best_particle_positions = bpp;
231 best_positions_initialized = true;
232 }
234 void setBestParticlePositions(std::vector<double> &&bpp) {
235 checkVarSize("ParticleSwarmState::setBestParticlePositions", "best particle positions", bpp.size(), num_dimensions * (num_particles + 1));
236 best_particle_positions = std::move(bpp);
237 best_positions_initialized = true;
238 }
239
241 void clearBestParticles() {
242 best_positions_initialized = false;
243 std::fill(best_particle_positions.begin(), best_particle_positions.end(), 0.0);
244 }
246 void clearCache() {
247 cache_initialized = false;
248 std::fill(cache_particle_fvals.begin(), cache_particle_fvals.end(), 0.0);
249 std::fill(cache_particle_inside.begin(), cache_particle_inside.end(), false);
250 std::fill(cache_best_particle_fvals.begin(), cache_best_particle_fvals.end(), 0.0);
251 std::fill(cache_best_particle_inside.begin(), cache_best_particle_inside.end(), false);
252 }
