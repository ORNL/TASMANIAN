103 GpuVector(GpuVector<T> &&other) : num_entries(std::exchange(other.num_entries, 0)), gpu_data(std::exchange(other.gpu_data, nullptr))

109 GpuVector<T>& operator =(GpuVector<T> &&other){

112 std::swap(gpu_data, temp.gpu_data);
113 #ifdef Tasmanian_ENABLE_DPCPP

175 template<typename U>
176 void load(AccelerationContext const *acc, size_t count, const U* cpu_data) {
177 if constexpr (std::is_same_v<U, T>) {
178 load_internal(acc, count, cpu_data);
179 } else {
180 // need to convert the array
181 std::vector<T> converted(count);
182 std::transform(cpu_data, cpu_data + count, converted.begin(), [](U const &x)->T{ return static_cast<T>(x); });
183 load_internal(acc, count, converted.data());
184 }
185 }
186
188 void unload(AccelerationContext const *acc, std::vector<T> &cpu_data) const{
189 cpu_data.resize(num_entries);
190 unload(acc, cpu_data.data());
191 }
193 std::vector<T> unload(AccelerationContext const *acc) const{
194 std::vector<T> y;
195 unload(acc, y);
196 return y;
197 }
199 void unload(AccelerationContext const *acc, size_t num, T* cpu_data) const;
201 void unload(AccelerationContext const *acc, T* cpu_data) const{ unload(acc, num_entries, cpu_data); }
202
204 T* eject(){
205 T* external = gpu_data;
206 gpu_data = nullptr;
207 num_entries = 0;
208 return external;
209 }

679 ChangeType favorSparse(bool favor){
680 AlgorithmPreference new_preference = [as=algorithm_select, favor]()->AlgorithmPreference{
681 if (favor){

683 }else{
684 return (as == algorithm_sparse) ? algorithm_autoselect : algorithm_dense;
685 }
686 }();

688 algorithm_select = new_preference;
689 return change_sparse_dense;
690 }else{
691 return change_none;
692 }

714 ChangeType testEnable(TypeAcceleration acc, int new_gpu_id) const{
715 TypeAcceleration effective_acc = AccelerationMeta::getAvailableFallback(acc);
716 #ifdef Tasmanian_ENABLE_DPCPP
717 if (AccelerationMeta::isAccTypeGPU(effective_acc) and ((new_gpu_id < -1 or new_gpu_id >= AccelerationMeta::getNumGpuDevices())))
718 throw std::runtime_error("Invalid GPU device ID, see ./tasgrid -v for list of detected devices.");
719 #else
720 if (AccelerationMeta::isAccTypeGPU(effective_acc) and ((new_gpu_id < 0 or new_gpu_id >= AccelerationMeta::getNumGpuDevices())))
721 throw std::runtime_error("Invalid GPU device ID, see ./tasgrid -v for list of detected devices.");
722 #endif
723 ChangeType mode_change = (effective_acc == mode) ? change_none : change_cpu_blas;
724 ChangeType device_change = (device == new_gpu_id) ? change_none : change_gpu_device;
725
726 if (on_gpu()){
727 return (AccelerationMeta::isAccTypeGPU(effective_acc)) ? device_change : change_gpu_device;
728 }else{
729 return (AccelerationMeta::isAccTypeGPU(effective_acc)) ? change_gpu_enabled : mode_change;
730 }
731 }

734 void enable(TypeAcceleration acc, int new_gpu_id){
735 // get the effective new acceleration mode (use the fallback if acc is not enabled)
736 TypeAcceleration effective_acc = AccelerationMeta::getAvailableFallback(acc);
737 // if switching to a GPU mode, check if the device id is valid
738 #ifdef Tasmanian_ENABLE_DPCPP
739 if (AccelerationMeta::isAccTypeGPU(effective_acc) and ((new_gpu_id < -1 or new_gpu_id >= AccelerationMeta::getNumGpuDevices())))
740 throw std::runtime_error("Invalid GPU device ID, see ./tasgrid -v for list of detected devices.");
741 #else
742 if (AccelerationMeta::isAccTypeGPU(effective_acc) and ((new_gpu_id < 0 or new_gpu_id >= AccelerationMeta::getNumGpuDevices())))
743 throw std::runtime_error("Invalid GPU device ID, see ./tasgrid -v for list of detected devices.");
744 #endif
745 if (AccelerationMeta::isAccTypeGPU(effective_acc)){
746 // if the new mode is GPU-based, make an engine or reset the engine if the device has changed
747 // if the engine exists and the device is not changed, then keep the existing engine
748 if (!engine or new_gpu_id != device)
749 engine.emplace();
750 }else{
751 engine.reset();
752 }
753
754 // assign the new values for the mode and device
755 mode = effective_acc;
756 device = new_gpu_id;
757 }
759 void setDevice() const{ AccelerationMeta::setDefaultGpuDevice(device); }
761 operator GpuEngine* () const{ return std::addressof(engine.value()); }
763 bool on_gpu() const{ return !!engine; }
